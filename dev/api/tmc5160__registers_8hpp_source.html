516[[nodiscard]] constexpr const RegInfo* FindReadable(uint8_t address) noexcept {
517 for (const RegInfo& r : kTable) {
518 if ((r.address == address) && !r.is_output && (r.access != Access::WriteOnly) &&
519 ((r.access != Access::Forbidden) || (address == kFactoryConf))) {
520 return &r;
521 }
522 }
523 return nullptr;
524}
525
531[[nodiscard]] constexpr const RegInfo* FindWritable(uint8_t address) noexcept {
532 for (const RegInfo& r : kTable) {
533 const bool writes = (r.access == Access::WriteOnly) || (r.access == Access::ReadWrite) ||
534 (r.access == Access::Clear);
535 if ((r.address == address) && writes) {
536 return &r;
537 }
538 }
539 return nullptr;
540}

547[[nodiscard]] constexpr uint32_t ResetValue(uint8_t address) noexcept {
548 for (const RegInfo& r : kTable) {
