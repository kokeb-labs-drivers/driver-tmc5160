149[[nodiscard]] constexpr uint8_t Crc8(std::span<const uint8_t> bytes) noexcept {
150 uint32_t crc = 0U;
151 for (const uint8_t byte : bytes) {
152 uint32_t current = byte;
153 for (uint32_t bit = 0U; bit < 8U; ++bit) {
154 const uint32_t feedback = ((crc >> 7U) ^ current) & 1U;
155 crc = (crc << 1U) & 0xFFU;
156 crc ^= feedback * 0x07U;
157 current >>= 1U;
158 }
159 }
160 return static_cast<uint8_t>(crc);
161}

173[[nodiscard]] constexpr UartFrame UartWriteFrame(uint8_t node, uint8_t reg,
174 uint32_t value) noexcept {
175 UartFrame f{kUartSync,
176 node,
177 static_cast<uint8_t>(0x80U | (static_cast<uint32_t>(reg) & 0x7FU)),
178 static_cast<uint8_t>(value >> 24U),
179 static_cast<uint8_t>(value >> 16U),

207[[nodiscard]] constexpr uint32_t ParseUartReply(const UartFrame& reply, uint8_t reg,
208 uint32_t& value) noexcept {
209 if ((reply[0] != kUartSync) || (reply[1] != kUartMasterAddress) ||
210 (reply[2] != (static_cast<uint32_t>(reg) & 0x7FU))) {
211 return 1U;
212 }
213 if (Crc8(std::span<const uint8_t>{reply}.first(7)) != reply[7]) {
214 return 2U;
215 }
216 value = (static_cast<uint32_t>(reply[3]) << 24U) | (static_cast<uint32_t>(reply[4]) << 16U) |
217 (static_cast<uint32_t>(reply[5]) << 8U) | static_cast<uint32_t>(reply[6]);
218 return 0U;
219}
