34[[nodiscard]] constexpr uint32_t RmsMa(uint32_t rsense_mohm, uint32_t global_scaler,
35 uint32_t cs_plus_1) noexcept {
36 if (rsense_mohm == 0U) {
37 return 0U;
38 }
39 const uint64_t num = uint64_t{kVfsMv} * 1000U * global_scaler * cs_plus_1 * kInvSqrt2Q16;
40 const uint64_t den = uint64_t{rsense_mohm} * 256U * 32U * 65'536U;
41 return static_cast<uint32_t>(num / den);
42}

58[[nodiscard]] constexpr uint32_t ScaleFor(uint32_t rsense_mohm, uint32_t gs,
59 uint32_t target_ma) noexcept {
60 uint32_t best = 0U;
61 for (uint32_t cs = 1U; cs <= 32U; ++cs) {
62 if (RmsMa(rsense_mohm, gs, cs) <= target_ma) {
63 best = cs;
64 }
65 }
66 return best;

93[[nodiscard]] constexpr Result<CurrentPlan> PlanCurrent(const Config& config, uint32_t run_ma,
94 uint32_t hold_ma) noexcept {
95 if ((run_ma == 0U) || (config.board.rsense_mohm == 0U)) {
96 return Result<CurrentPlan>::Failure(Error::InvalidArgument);
97 }
98 if ((hold_ma > run_ma) && !config.ack.Has(Ack::HoldAboveRun)) {
99 return Result<CurrentPlan>::Failure(Error::Refused);
100 }
101 const uint32_t ceiling = detail::Ceiling(config);
102 const uint32_t highest = (hold_ma > run_ma) ? hold_ma : run_ma;
103 const uint32_t fs256 = RmsMa(config.board.rsense_mohm, 256U, 32U);
104 if ((highest > ceiling) || (highest > fs256)) {
105 return Result<CurrentPlan>::Failure(Error::LimitExceeded);
106 }
107 uint32_t gs = 129U;
108 while ((gs < 256U) && (RmsMa(config.board.rsense_mohm, gs, 32U) < highest)) {
109 ++gs;
110 }
111 const uint32_t cs_run = detail::ScaleFor(config.board.rsense_mohm, gs, run_ma);
112 const uint32_t cs_hold = detail::ScaleFor(config.board.rsense_mohm, gs, hold_ma);
113 if (cs_run == 0U) {
114 return Result<CurrentPlan>::Failure(Error::LimitExceeded);
115 }
116 CurrentPlan p{};
117 p.global_scaler = gs;
118 p.global_scaler_reg = (gs == 256U) ? 0U : gs;
119 p.irun = cs_run - 1U;
120 p.ihold = (cs_hold == 0U) ? 0U : (cs_hold - 1U);
121 p.run_ma = RmsMa(config.board.rsense_mohm, gs, cs_run);
122 p.hold_ma = RmsMa(config.board.rsense_mohm, gs, p.ihold + 1U);
123 p.full_scale_ma = RmsMa(config.board.rsense_mohm, gs, 32U);
124 return Result<CurrentPlan>::Success(p);
125}

359[[nodiscard]] constexpr Result<EncoderScale> EncoderScaleFor(uint32_t microsteps, uint32_t counts,
360 bool decimal, bool negative) noexcept {
361 if ((microsteps == 0U) || (counts == 0U) || ((microsteps / counts) > 32'767U)) {
362 return Result<EncoderScale>::Failure(Error::InvalidArgument);
363 }
364 const uint64_t unit = decimal ? 10'000U : 65'536U;
365 const uint64_t scaled = (uint64_t{microsteps} * unit) / counts;
366 EncoderScale s{};
367 s.decimal = decimal;
368 s.integer = static_cast<int16_t>(scaled / unit);
369 s.fraction = static_cast<uint16_t>(scaled % unit);
370 s.negative = negative;
371 return Result<EncoderScale>::Success(s);

382[[nodiscard]] constexpr uint32_t EncConstWord(const EncoderScale& scale) noexcept {
383 if (!scale.negative) {
384 return (static_cast<uint32_t>(static_cast<uint16_t>(scale.integer)) << 16U) | scale.fraction;
385 }
386 if (scale.decimal) {
387 const uint32_t frac = (scale.fraction == 0U) ? 0U : (10'000U - scale.fraction);
388 const int32_t whole = (scale.fraction == 0U) ? -static_cast<int32_t>(scale.integer)
389 : -static_cast<int32_t>(scale.integer) - 1;
390 return (static_cast<uint32_t>(static_cast<uint16_t>(static_cast<int16_t>(whole))) << 16U) |
391 frac;
392 }
393 const uint32_t magnitude =
394 (static_cast<uint32_t>(static_cast<uint16_t>(scale.integer)) << 16U) | scale.fraction;
395 return 0U - magnitude;
396}
