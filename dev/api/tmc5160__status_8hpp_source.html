149[[nodiscard]] constexpr int32_t SignExtend(uint32_t v, uint32_t bits) noexcept {
150 const uint32_t sign = uint32_t{1} << (bits - 1U);
151 const uint32_t low = v & ((sign << 1U) - 1U);
152 return static_cast<int32_t>((low ^ sign)) - static_cast<int32_t>(sign);
153}

244[[nodiscard]] constexpr uint32_t GconfWord(const Config& c, bool direct) noexcept {
245 const Diag& d = c.diag;
246 uint32_t w = Put(c.flags.recalibrate, 0) | Put(c.standstill.fast_standstill, 1) |
247 Put(c.stealth.enable, 2) | Put(c.flags.multistep_filt, 3) |
248 Put(c.flags.invert_direction, 4);
249 w |= Put(d.diag0_error, 5) | Put(d.diag0_otpw, 6) | Put(d.diag0_stall || d.diag0_step_output, 7);
250 w |= Put(d.diag1_stall || d.diag1_dir_output, 8) | Put(d.diag1_index, 9) |
251 Put(d.diag1_on_state, 10);
252 w |= Put(d.diag1_steps_skipped, 11) | Put(d.diag0_push_pull, 12) | Put(d.diag1_push_pull, 13);
253 w |= Put(c.thresholds.small_hysteresis, 14) | Put(c.flags.stop_input, 15) | Put(direct, 16);
254 return w;
255}

258[[nodiscard]] constexpr uint32_t ChopconfWord(const Config& c, uint32_t toff) noexcept {
259 const Chopper& k = c.chopper;
260 const bool const_off = k.mode == ChopperMode::ConstOffTime;
261 const uint32_t low = const_off ? (k.tfd & 7U) : k.hstrt;
262 const uint32_t high = const_off ? k.offset : k.hend;
263 uint32_t w = toff & 15U;
264 w |= low << 4U;
265 w |= high << 7U;
266 w |= Put(const_off && ((k.tfd & 8U) != 0U), 11) | Put(const_off && k.no_comparator, 12);
267 w |= Put(const_off, 14) | (static_cast<uint32_t>(k.tbl & 3U) << 15U);
268 w |= Put(k.high_fullstep, 18) | Put(k.high_const_off, 19) |
269 (static_cast<uint32_t>(k.tpfd & 15U) << 20U);
270 w |= (static_cast<uint32_t>(k.mres) << 24U) | Put(k.interpolate, 28) | Put(k.double_edge, 29);
271 w |= Put(c.protection.disable_s2g, 30) | Put(c.protection.disable_s2vs, 31);
272 return w;
273}

471[[nodiscard]] constexpr uint32_t WordFor(const WordContext& x, uint8_t address) noexcept {
472 for (const WordEntry& e : kWordTable) {
473 if (e.address == address) {
474 return e.encode(x);
