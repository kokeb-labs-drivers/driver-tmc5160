622[[nodiscard]] constexpr int32_t TableStep(uint32_t width_code, uint32_t bit) noexcept {
623 return static_cast<int32_t>(width_code & 3U) - 1 + static_cast<int32_t>(bit & 1U);
624}

627[[nodiscard]] constexpr uint32_t SegmentWidth(const MicrostepTable& t, uint32_t n) noexcept {
628 if (n < t.x1) {
629 return t.w[0];
630 }
631 if (n < t.x2) {
632 return t.w[1];
633 }
634 return (n < t.x3) ? t.w[2] : t.w[3];
635}

638[[nodiscard]] constexpr int32_t TableEntry(const MicrostepTable& t, uint32_t n) noexcept {
639 auto value = static_cast<int32_t>(t.start_sin);
640 for (uint32_t i = 1U; i <= (n & 255U); ++i) {
641 const uint32_t bit = (t.words[(i >> 5U) & 7U] >> (i & 31U)) & 1U;
642 value += TableStep(SegmentWidth(t, i), bit);
643 }
644 return value;
645}

654[[nodiscard]] constexpr Error CheckTable(const MicrostepTable& table) noexcept {
655 if ((table.x1 == 0U) || (table.x1 >= table.x2) || (table.x2 > table.x3)) {
656 return Error::InvalidArgument;
657 }
658 auto value = static_cast<int32_t>(table.start_sin);
659 for (uint32_t i = 1U; i <= 255U; ++i) {
660 const uint32_t bit = (table.words[(i >> 5U) & 7U] >> (i & 31U)) & 1U;
661 value += detail::TableStep(detail::SegmentWidth(table, i), bit);
662 if ((value < -248) || (value > 248)) {
663 return Error::InvalidArgument;
664 }
665 }
666 const auto last = static_cast<int32_t>(table.start_sin90);
667 return (value == last) ? Error::Ok : Error::InvalidArgument;

849[[nodiscard]] constexpr Error CheckAcks(const Config& c) noexcept {
850 using detail::Both;
851 using detail::NeedAck;
852 const bool short_off = c.protection.disable_s2g || c.protection.disable_s2vs;
853 const bool low_ot = (c.bridge.ot_level == OtLevel::C120) ||
854 ((c.bridge.ot_level == OtLevel::C136) && (c.board.supply_mv > 24'000U));
855 const bool hard_stop =

857 !c.switches.soft_stop;
858 Error e = NeedAck(short_off, c, Ack::ShortProtectionOff);
859 e = Both(e, NeedAck(c.bridge.declared && low_ot, c, Ack::LowOtLevel));
860 e = Both(e, NeedAck(hard_stop && c.stealth.enable, c, Ack::HardStopWithStealthChop));
861 return Both(e, NeedAck(c.ramp.vstart > c.ramp.vstop, c, Ack::VstartAboveVstop));
862}
